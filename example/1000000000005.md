## Go

```go
package main

import (
    "fmt"
    "time"
)

// Greeting は挨拶メッセージを生成する。
type Greeting struct {
    Name string
}

func (g *Greeting) Say() string {
    return fmt.Sprintf("Hello, %s! It's %s.", g.Name, time.Now().Format("15:04"))
}

func main() {
    g := &Greeting{Name: "ivvvy"}
    fmt.Println(g.Say())
}
```

## JavaScript

```javascript
// 検索インデックスを取得して絞り込む
async function search(query) {
    const response = await fetch('/search/index.json');
    const index = await response.json();

    const terms = query.toLowerCase().split(/\s+/);

    return index.pages.filter(page => {
        return terms.every(term =>
            page.tokens.includes(term) ||
            page.title.toLowerCase().includes(term)
        );
    });
}

search('markdown callout').then(results => {
    results.forEach(r => console.log(r.title));
});
```

## SQL

```sql
-- ページの全文検索（参考：実際はクライアントサイドで実行）
SELECT
    p.id,
    p.title,
    p.body,
    p.updated_at
FROM pages p
WHERE
    p.body LIKE '%Markdown%'
    AND p.updated_at >= '2024-01-01'
ORDER BY p.updated_at DESC
LIMIT 20;
```

## シェルスクリプト（Bash）

```bash
#!/bin/bash
set -euo pipefail

# ivvvy をビルドして起動する
BUILD_DIR="./dist"
DATA_DIR="./data"

echo "Building ivvvy..."
go build -o "${BUILD_DIR}/ivvvy" .

echo "Starting server on port ${PORT}..."
"${BUILD_DIR}/ivvvy" \
    -port "${PORT}" \
    -data "${DATA_DIR}"
```
